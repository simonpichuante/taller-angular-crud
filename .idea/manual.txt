import { LogService } from './log.service';
@Injectable()
export class ErrorGlobalService implements ErrorHandler {

  private alerta = inject(AlertaService);
  handleError(error: Error): void {
    this.log.error(`Error no controlado: ${error.message}`);
    this.alerta.error('Ocurrió un error inesperado.');
  }
}

// ============================================================
// alerta.service.ts
// Muestra mensajes en pantalla desde cualquier parte de la app.
// Uso: this.alerta.exito('Empleado guardado');
// El componente <app-alertas> se encarga de dibujarlos.
// ============================================================
import { Injectable, signal } from '@angular/core';
import { Alerta } from '../../models/alerta.model';
@Injectable({ providedIn: 'root' })
export class AlertaService {
  alertas = signal<Alerta[]>([]);
  private siguienteId = 1;
  exito(mensaje: string): void {
    this.mostrar('exito', mensaje);
  }
  error(mensaje: string): void {
    this.mostrar('error', mensaje);
  }
  advertencia(mensaje: string): void {
    this.mostrar('advertencia', mensaje);
  }
  info(mensaje: string): void {
    this.mostrar('info', mensaje);
  }
  cerrar(id: number): void {
Control de Empleados · Frontend Angular · Semana 7 15 / 49
    this.alertas.update((lista) => lista.filter((alerta) => alerta.id !== id));
  }
  private mostrar(tipo: Alerta['tipo'], mensaje: string): void {
    // Si el mismo mensaje ya está en pantalla, no se repite.
    if (this.alertas().some((alerta) => alerta.mensaje === mensaje)) {
      return;
    }
    const id = this.siguienteId++;
    this.alertas.update((lista) => [...lista, { id, tipo, mensaje }]);
    // Después de 5 segundos se cierra sola.
    setTimeout(() => this.cerrar(id), 5000);
  }
}
Cómo
funciona
1. Hay un método por cada tipo: exito , error , advertencia  e info . Los cuatro llaman a mostrar() .
2. Si el mismo mensaje ya está en pantalla, no se repite. Esto evita, por ejemplo, cuatro avisos iguales
cuando la API está apagada.
3. Cada alerta recibe un id  para poder cerrarla.
4. setTimeout  la cierra sola después de 5 segundos. También se puede cerrar con la ×.

// ============================================================
// carga.service.ts
// Controla la pantalla de carga (capa oscura con spinner).
// Usa un contador: con 3 peticiones a la vez, la pantalla sigue
// bloqueada hasta que termine la última.
// ============================================================
import { Injectable, computed, signal } from '@angular/core';
@Injectable({ providedIn: 'root' })
export class CargaService {
  private peticiones = signal(0);
  // true mientras exista al menos una petición en curso.
  activa = computed(() => this.peticiones() > 0);
  mostrar(): void {
    this.peticiones.update((n) => n + 1);
  }
  ocultar(): void {
    this.peticiones.update((n) => n - 1);
  }
}
¿Por

y

La página de empleados hace 4 peticiones al abrirse. Si se usara true/false , la primera en terminar

petición suma 1 al empezar y resta 1 al terminar. El spinner se va recién cuando el contador vuelve a 0.
computed  crea un signal que se calcula a partir de otro: activa  vale true  mientras peticiones  sea
mayor que 0.

// Único lugar que habla con la API. Cada petición pasa por aquí,
// y por eso todas muestran la pantalla de carga y quedan en el log.
// Los demás servicios de esta carpeta lo usan así:

import { HttpClient } from '@angular/common/http';
import { Injectable, inject } from '@angular/core';
import { Observable, finalize, tap } from 'rxjs';
import { environment } from '../../../environments/environment';
import { LogService } from '../sistema/log.service';
import { CargaService } from '../ui/carga.service';
@Injectable({ providedIn: 'root' })
export class ApiService {
  private http = inject(HttpClient);
  private carga = inject(CargaService);
  private log = inject(LogService);
  private url = environment.apiUrl;
  get<T>(ruta: string, params: Record<string, string> = {}): Observable<T> {
    return this.ejecutar(`GET ${ruta}`, this.http.get<T>(this.url + ruta, { params }));
  }
  post<T>(ruta: string, datos: unknown): Observable<T> {
    return this.ejecutar(`POST ${ruta}`, this.http.post<T>(this.url + ruta, datos));
  }
  put<T>(ruta: string, datos: unknown): Observable<T> {
    return this.ejecutar(`PUT ${ruta}`, this.http.put<T>(this.url + ruta, datos));
  }
  delete<T>(ruta: string): Observable<T> {
    return this.ejecutar(`DELETE ${ruta}`, this.http.delete<T>(this.url + ruta));
  }
  private ejecutar<T>(descripcion: string, peticion: Observable<T>): Observable<T> {
    this.carga.mostrar();
    return peticion.pipe(
      // tap permite mirar el resultado sin modificarlo.
      tap({
Control de Empleados · Frontend Angular · Semana 7 23 / 49
        next: () => this.log.info(`${descripcion}: correcto`),
        error: (error) => this.log.error(`${descripcion}: error ${error.status}`),
      }),
      // finalize se ejecuta siempre: si la petición sale bien o si falla.
      finalize(() => this.carga.ocultar()),
    );
  }

Cómo
funciona
1. get , post , put  y delete  arman la URL completa y llaman a ejecutar() .
2. ejecutar()  enciende la carga con this.carga.mostrar() .
3. pipe()  agrega pasos a la respuesta. tap  la mira sin cambiarla y guarda en el log si salió bien o mal.
4. finalize  se ejecuta siempre, salga bien o mal, y apaga la carga. Así la pantalla nunca queda
bloqueada por un error.
Regla
del
proyecto: toda petición a la API debe pasar por ApiService . Si un servicio usa HttpClient
directamente, esa petición no tendrá spinner ni log.

// ============================================================
// mensaje-error.ts
// Convierte un error de la API en un texto para el usuario.
// Uso: this.alerta.error(mensajeError(error));
// ============================================================
import { HttpErrorResponse } from '@angular/common/http';
export function mensajeError(error: HttpErrorResponse): string {
  const detalle = error.error?.detail;
  if (error.status === 0) {
    return 'No se pudo conectar con la API. Revisa que esté encendida.';
  }
  // 400, 404 y 409: la API ya envía el mensaje en español.
  if (typeof detalle === 'string') {
    return detalle;
  }
  // 422: la API envía una lista de errores. Se muestra el primero.
  if (Array.isArray(detalle)) {
    return detalle[0].msg.replace('Value error, ', '');
  }
  return 'Ocurrió un error en el servidor.';
}

Selectores
en
cascada
Las comunas dependen de la provincia, y las provincias dependen de la región:
1. Al abrir la página se cargan las 16 regiones.
2. Al elegir una región, cambiarRegion()  limpia la provincia y la comuna, y pide las provincias de esa
región.
3. Al elegir una provincia, cambiarProvincia()  limpia la comuna y pide sus comunas.
4. Solo comuna_id  se envía a la API: la región y la provincia se deducen de la comuna.

15.
Cómo
seguir
ampliando
el
proyecto
Para agregar un mantenedor nuevo, por ejemplo Sucursales, se repiten los mismos pasos:
1. Crear la tabla y los endpoints en el backend.
2. Crear el modelo y escribir en él SucursalForm  y Sucursal .
3. Crear el servicio y escribir sus métodos usando ApiService .
4. Crear la página y completarla copiando el patrón de Departamentos.
5. Agregar la ruta en app.routes.ts  y el enlace en components/menu/menu.html .
Terminal (carpeta frontend)
npx ng g interface models/sucursal --type=model
npx ng g service services/api/sucursal --type=service --skip-tests
npx ng g component pages/sucursales --skip-tests --style=none
Al usar ApiService , AlertaService  y LogService , la página nueva tiene desde el comienzo pantalla de
carga, alertas y registro de eventos.
